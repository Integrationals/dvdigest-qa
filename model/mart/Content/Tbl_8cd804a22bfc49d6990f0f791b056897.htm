PreSQL   
DDL Code  CREATE TRANSIENT TABLE FINDING_FACT(
     FINDING_KEY       VARCHAR(128)          NOT NULL,
     SOURCE_KEY        VARCHAR(16777216),
     DETECTED_DATE     DATE,
     FINDER_NAME       VARCHAR(128),
     FINDER_TYPE       VARCHAR(32),
     SOURCE_TABLE      VARCHAR(256),
     SOURCE_COLUMN     VARCHAR(128),
     COLUMN_VALUE      VARCHAR(512),
     STATUS            VARCHAR(32),
     FINDING_KIND      VARCHAR(16777216),
     FINDING_ROW       VARCHAR(770),
     CONSTRAINT PK_FINDING_FACT PRIMARY KEY (FINDING_KEY),
     CONSTRAINT FK_FINDING_FACT_DETECTED_DATE FOREIGN KEY (DETECTED_DATE)
     REFERENCES DAY_DIM(DATE_DAY),
     CONSTRAINT FK_FINDING_FACT_SOURCE_KEY FOREIGN KEY (SOURCE_KEY)
     REFERENCES SOURCE_DIM(SOURCE_KEY)
)
;

ColumnName  Domain  Datatype  NULL  Definition 
FINDING_KEY    VARCHAR(128)  NO  
The finding''s key in CONFIG.DATA_FINDING.
 
SOURCE_KEY    VARCHAR(16777216)  YES  
The source the flagged record came from, a source_dim key (SSI or NFL).
 
DETECTED_DATE    DATE  YES  
The UTC day the finding was written.
 
FINDER_NAME    VARCHAR(128)  YES  
The inspector or auditor that wrote the finding.
 
FINDER_TYPE    VARCHAR(32)  YES  
inspection or audit.
 
SOURCE_TABLE    VARCHAR(256)  YES  
The landing table of the flagged record.
 
SOURCE_COLUMN    VARCHAR(128)  YES  
The column that failed.
 
COLUMN_VALUE    VARCHAR(512)  YES  
The flagged value, or the name of the row when its key is missing.
 
STATUS    VARCHAR(32)  YES  
The finding''s status: detected until someone acknowledges it.
 
FINDING_KIND    VARCHAR(16777216)  YES  
form, the record arrived in a form the vault does not accept; missing_key, it arrived without a business key and was kept and flagged; audit, a count that does not reconcile.
 
FINDING_ROW    VARCHAR(770)  YES  
The row a finding names, its table and the row''s name, so a row missing two keys counts once.
