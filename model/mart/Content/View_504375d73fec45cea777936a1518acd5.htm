DDL Code  create or replace view CARRIER_DIM(
      CARRIER_KEY COMMENT 'The carrier''s key, an organization_dim key.',
      CARRIER_CODE COMMENT 'The carrier code.',
      CARRIER_NAME COMMENT 'The carrier''s name, as it sends it.'
) COMMENT='One carrier, a role view over organization_dim for the organizations sent through carrier feeds.'
as (

SELECT organization_key AS carrier_key, organization_code AS carrier_code, organization_name AS carrier_name
FROM DV.CONFORMED_DM.organization_dim
WHERE source_key IN ('NFL', 'PMC')         -- organizations sent through a carrier feed
    OR source_key IN ('SYSTEM', 'ERROR')    -- the two ghost rows, mart design D-14
   )
