DDL Code  create or replace view CLIENT_DIM(
      CLIENT_KEY COMMENT 'The client''s key, a person_dim key.',
      CLIENT_NUMBER COMMENT 'Sunbelt Senior''s client number.',
      FIRST_NAME COMMENT 'The client''s first name.',
      LAST_NAME COMMENT 'The client''s last name.',
      STATE COMMENT 'The client''s state.',
      SOURCE_KEY COMMENT 'The source that first sent the client.',
      FIRST_KNOWN_DATE COMMENT 'The day the vault first received the client.'
) COMMENT='One client record of Sunbelt Senior, a role view over person_dim, with the two ghost rows (D-14).'
as (

SELECT person_key AS client_key, person_number AS client_number, first_name, last_name, state,
        source_key, first_known_date
FROM DV.CONFORMED_DM.person_dim
WHERE person_role = 'client'
    OR source_key IN ('SYSTEM', 'ERROR')    -- the two ghost rows, mart design D-14
   )
