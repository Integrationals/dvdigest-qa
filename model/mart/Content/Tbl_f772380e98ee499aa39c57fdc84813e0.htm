Definition  
One version of a carrier''s terms for one policy, from one load until the next. Serves Q-10 and Q-14.
 
Note 
 
PreSQL   
DDL Code  CREATE TRANSIENT TABLE POLICY_TERMS_FACT(
     POLICY_TERMS_KEY     VARCHAR(32)           NOT NULL,
     POLICY_KEY           VARCHAR(16777216),
     CARRIER_KEY          VARCHAR(16777216),
     VALID_FROM_DATE      DATE,
     VALID_TO_DATE        DATE,
     EFFECTIVE_DATE       DATE,
     FACE_AMOUNT          NUMBER(12, 2),
     STATUS               VARCHAR(9),
     CONSTRAINT PK_POLICY_TERMS_FACT PRIMARY KEY (POLICY_TERMS_KEY),
     CONSTRAINT FK_POLICY_TERMS_FACT_POLICY_KEY FOREIGN KEY (POLICY_KEY)
     REFERENCES POLICY_DIM(POLICY_KEY),
     CONSTRAINT FK_POLICY_TERMS_FACT_VALID_FROM_DATE FOREIGN KEY (VALID_FROM_DATE)
     REFERENCES DAY_DIM(DATE_DAY),
     CONSTRAINT FK_POLICY_TERMS_FACT_VALID_TO_DATE FOREIGN KEY (VALID_TO_DATE)
     REFERENCES DAY_DIM(DATE_DAY),
     CONSTRAINT FK_POLICY_TERMS_FACT_CARRIER_KEY FOREIGN KEY (CARRIER_KEY)
     REFERENCES ORGANIZATION_DIM(ORGANIZATION_KEY),
     CONSTRAINT FK_POLICY_TERMS_FACT_EFFECTIVE_DATE FOREIGN KEY (EFFECTIVE_DATE)
     REFERENCES DAY_DIM(DATE_DAY)
)
;

POLICY_TERMS_FACT Columns 
ColumnName  Domain  Datatype  NULL  Definition 
POLICY_TERMS_KEY    VARCHAR(32)  NO  
The row''s key, a hash of the policy key and the version''s load.
 
POLICY_KEY    VARCHAR(16777216)  YES  
The policy, a policy_dim key.
 
CARRIER_KEY    VARCHAR(16777216)  YES  
The carrier that issued the policy, a foreign key to organization_dim; the unknown organization when the policy''s carrier code names no organization record (mart design D-17).
 
VALID_FROM_DATE    DATE  YES  
The day this version arrived.
 
VALID_TO_DATE    DATE  YES  
The last day this version held; empty while it is the current version.
 
EFFECTIVE_DATE    DATE  YES  
The day the carrier says the policy took effect.
 
FACE_AMOUNT    NUMBER(12, 2)  YES  
The face amount the carrier states, in US dollars as sent; empty when not sent.
 
STATUS    VARCHAR(9)  YES  
The policy''s status in this version, as the carrier sends it (IF in force, LP lapsed).
