DDL Code  create or replace view FINDING_FACT(
      FINDING_KEY COMMENT 'The finding''s key in CONFIG.DATA_FINDING.',
      SOURCE_KEY COMMENT 'The source the flagged record came from, a source_dim key (SSI or NFL).',
      DETECTED_DATE COMMENT 'The UTC day the finding was written.',
      FINDER_NAME COMMENT 'The inspector or auditor that wrote the finding.',
      FINDER_TYPE COMMENT 'inspection or audit.',
      SOURCE_TABLE COMMENT 'The landing table of the flagged record.',
      SOURCE_COLUMN COMMENT 'The column that failed.',
      COLUMN_VALUE COMMENT 'The flagged value, or the name of the row when its key is missing.',
      STATUS COMMENT 'The finding''s status: detected until someone acknowledges it.',
      FINDING_KIND COMMENT 'form, the record arrived in a form the vault does not accept; missing_key, it arrived without a business key and was kept and flagged; audit, a count that does not reconcile.',
      FINDING_ROW COMMENT 'The row a finding names, its table and the row''s name, so a row missing two keys counts once.'
) COMMENT='One record an inspector or auditor flagged in the landing''s CONFIG.DATA_FINDING. Serves Q-15.'
as (


WITH f AS (
SELECT finding_pk AS finding_key,
        CAST(CASE source_system WHEN 'SUNBELT' THEN 'SSI' WHEN 'NORTHFIELD' THEN 'NFL' ELSE source_system END AS VARCHAR) AS source_key,   -- source_dim's type, so the foreign key can be declared (guide step 82)
        TO_DATE(CONVERT_TIMEZONE('UTC', detected_at)) AS detected_date,
        finder_name, finder_type, source_table, source_column, column_value, status,
        CASE WHEN finder_type = 'audit' THEN 'audit'
             WHEN ENDSWITH(finder_name, '_key_missing') THEN 'missing_key'
             WHEN ENDSWITH(finder_name, '_form') THEN 'form' END AS finding_kind,
        source_table || ': ' || column_value AS finding_row
FROM LANDING.CONFIG.data_finding
)

SELECT f.finding_key, COALESCE(s.source_key, f.source_key) AS source_key, COALESCE(d.date_day, f.detected_date) AS detected_date,
        f.finder_name, f.finder_type, f.source_table, f.source_column, f.column_value, f.status, f.finding_kind, f.finding_row
FROM f
LEFT JOIN DV.CONFORMED_DM.source_dim s ON s.source_key = f.source_key
LEFT JOIN DV.CONFORMED_DM.day_dim d ON d.date_day = f.detected_date
   )
