DDL Code  create or replace secure view INSURED_DIM(
      INSURED_KEY COMMENT 'The insured''s key, a person_dim key.',
      FULL_NAME COMMENT 'The insured''s name, as Northfield Life sends it.',
      BIRTH_DATE COMMENT 'The birth year only, unless the session holds SYSADMIN.',
      ADDRESS COMMENT 'The state only, unless the session holds SYSADMIN.'
) COMMENT='One insured record of Northfield Life, a role view over person_dim; birth_date and address masked except for SYSADMIN (mart design D-9).'
as (


SELECT person_key AS insured_key, full_name,
        CASE WHEN IS_ROLE_IN_SESSION('SYSADMIN') THEN birth_date ELSE LEFT(birth_date, 4) END AS birth_date,
        CASE WHEN IS_ROLE_IN_SESSION('SYSADMIN') THEN address ELSE RIGHT(TRIM(address), 2) END AS address
FROM DV.CONFORMED_DM.person_dim
WHERE person_role = 'insured'
    OR source_key IN ('SYSTEM', 'ERROR')    -- the two ghost rows, mart design D-14
   )
