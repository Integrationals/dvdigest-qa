{#
  Layer 6 auditor (design D-10; guide step 31). Compares one system against another: every distinct usable policy key staged from NORTHFIELD.POLICY must be in
  the policy hub, whichever record source the hub kept (path PH3: a key both sources send in one load is kept
  under one of them). A first version compared the hub rows of its own record source and raised a false
  finding at act 1 (2026-10-25); reduced to unit tests A4 to A6, red, then fixed.
  Its disagree test was red 2026-10-25 01:12 UTC on beb49c9 before the comparison, green with it. Writes one summary finding, carrying both counts, only when they disagree.
  Pattern: a reconciliation writes at most one finding.
#}
{{ config(
    materialized='merge_into',
    meta={
        'target_database': landing_storage(),
        'target_schema': 'CONFIG',
        'target_table': 'DATA_FINDING',
        'unique_key': 'FINDING_PK',
        'diff_key': 'FINDING_PK',
        'ts_key': 'DETECTED_AT'
    }
) }}

WITH source_keys AS (
    SELECT DISTINCT hk_policy_h
    FROM {{ ref('stg_northfield_policy') }}
    WHERE hk_policy_h NOT IN ('00000000000000000000000000000000', 'ffffffffffffffffffffffffffffffff')
),

counts AS (
    SELECT COUNT(*) AS n, COUNT(h.hk_policy_h) AS found
    FROM source_keys k
    LEFT JOIN (SELECT DISTINCT hk_policy_h FROM {{ ref('policy_h') }}) h ON h.hk_policy_h = k.hk_policy_h
)

SELECT
    {{ dbt_utils.generate_surrogate_key(["'auditor_northfield_policy_hub'", "'NORTHFIELD'", "'NORTHFIELD.POLICY'"]) }} AS finding_pk,
    'NORTHFIELD' AS source_system,
    'NORTHFIELD.POLICY' AS source_table,
    'POLICY_NUMBER' AS source_column,
    c.n || ' source keys, ' || c.found || ' found in the hub' AS column_value,
    'auditor_northfield_policy_hub' AS finder_name,
    'audit' AS finder_type,
    'Not every policy key of NORTHFIELD.POLICY is in the policy hub: ' || c.n || ' distinct usable keys staged, ' || c.found || ' found in policy_h.' AS detail,
    'detected' AS status,
    CAST(NULL AS VARCHAR(32)) AS finding_id,
    CAST(NULL AS VARCHAR(1024)) AS notes,
    CAST(NULL AS VARCHAR(2048)) AS root_cause,
    CURRENT_TIMESTAMP() AS detected_at,
    CAST(NULL AS VARCHAR(64)) AS acknowledged_by,
    CAST(NULL AS TIMESTAMP_TZ) AS acknowledged_at
FROM counts c
WHERE c.n <> c.found

dbt/models/inspectors/inspector_northfield_policy_key_missing.sql
{#
  Layer 5 inspector (fix to release 1, 2026-10-28; Code Review Standard k12ki19: no silent record loss).
  Discloses each row of LANDING.NORTHFIELD.POLICY, the persistent staging that keeps every row as sent, that arrived without a
  policy number, an insured id or a writing agent's number, missing or blank by staging's rule NULLIF(TRIM(...), ''). Staging
  maps each such key to the zero key and the satellites keep the row out (design D-44), so the finding keeps it in view (D-45). One finding per row and missing
  key (POLICY_NUMBER, INSURED_ID, WRITING_AGENT_NPN); the column value names the row so it can be found in landing storage. Never filters or corrects.
#}
{{ config(
    materialized='merge_into',
    meta={
        'target_database': landing_storage(),
        'target_schema': 'CONFIG',
        'target_table': 'DATA_FINDING',
        'unique_key': 'FINDING_PK',
        'diff_key': 'FINDING_PK',
        'ts_key': 'DETECTED_AT'
    }
) }}

WITH policies AS (
    SELECT NULLIF(TRIM(policy_number), '') AS policy_number, NULLIF(TRIM(insured_id), '') AS insured_number,
           NULLIF(TRIM(writing_agent_npn), '') AS agent_npn,
           'Policy row ' || COALESCE(NULLIF(TRIM(policy_number), ''), '(no policy number)') || ', loaded '
               || TO_VARCHAR(TO_TIMESTAMP_NTZ(loaded_at), 'YYYY-MM-DD') AS row_name
    FROM {{ source('northfield', 'policy') }}
),
missing AS (
    SELECT 'POLICY_NUMBER' AS source_column, row_name AS column_value,
           'Policy row without a policy number: it stays out of the policy satellite.' AS detail
    FROM policies WHERE policy_number IS NULL
    UNION ALL
    SELECT 'INSURED_ID', row_name, 'Policy row without an insured id: it stays out of the insured satellite, and its policy links to the unknown insured.' FROM policies WHERE insured_number IS NULL
    UNION ALL
    SELECT 'WRITING_AGENT_NPN', row_name, 'Policy row without a writing agent''s number: its policy links to the unknown agent.' FROM policies WHERE agent_npn IS NULL
)

SELECT
    {{ dbt_utils.generate_surrogate_key(["'inspector_northfield_policy_key_missing'", "'NORTHFIELD'", "'NORTHFIELD.POLICY'", "source_column", "column_value"]) }} AS finding_pk,
    'NORTHFIELD' AS source_system,
    'NORTHFIELD.POLICY' AS source_table,
    source_column,
    LEFT(column_value, 500) AS column_value,
    'inspector_northfield_policy_key_missing' AS finder_name,
    'inspection' AS finder_type,
    detail,
    'detected' AS status,
    CAST(NULL AS VARCHAR(32)) AS finding_id,
    CAST(NULL AS VARCHAR(1024)) AS notes,
    CAST(NULL AS VARCHAR(2048)) AS root_cause,
    CURRENT_TIMESTAMP() AS detected_at,
    CAST(NULL AS VARCHAR(64)) AS acknowledged_by,
    CAST(NULL AS TIMESTAMP_TZ) AS acknowledged_at
FROM missing
