logger = logging.getLogger()
logger.setLevel(logging.INFO)
STARTED = time.perf_counter()
COLD = True


async def send_response(send, status: int, body: dict, content_type="application/json") -> None:
    payload = response(status, body)
    payload["headers"]["Content-Type"] = content_type
    await send({
        "type": "http.response.start", "status": status,
        "headers": [(key.lower().encode(), value.encode()) for key, value in payload["headers"].items()],
    })
    await send({"type": "http.response.body", "body": payload["body"].encode()})


def log_start(question: str, streaming: bool) -> None:
    global COLD
    logger.info(json.dumps({
        "event": "conversation_start", "environment": environment(), "question": question,
        "streaming": streaming, "cold_start": COLD,
        "since_import_ms": round((time.perf_counter() - STARTED) * 1000) if COLD else 0,
    }))
    COLD = False


async def app(scope, receive, send) -> None:
    if scope["type"] != "http":
        return
    method = scope["method"]
    if method == "GET":
        await send_response(send, 200, {"status": "ok"})
        return
    if method == "OPTIONS":
        await send_response(send, 204, {})
        return
    if method != "POST":
        await send_response(send, 405, {"error": "invalid_request"})
        return
    body = b""
    while True:
        message = await receive()
        body += message.get("body", b"")
        if not message.get("more_body"):
            break
    try:
        value = json.loads(body or b"{}")
        if not isinstance(value, dict):
            raise ValueError("invalid_request")
    except (ValueError, UnicodeDecodeError):
        await send_response(send, 400, {"error": "invalid_request"})
        return
    if not valid_access_code(value.get("access_code")):
        await send_response(send, 401, {"error": "access_code"})
        return
    try:
        messages = validate_messages(value.get("messages"))
    except ValueError as error:
        await send_response(send, 400, {"error": str(error)})
        return
    if not consume_daily_allowance("conversation", int(os.environ.get("DIGEST_DAILY_CAP", "200"))):
        await send_response(send, 429, {"error": "daily_cap"})
        return
    question = next((item["content"] for item in reversed(messages) if item["role"] == "user"), "")
    headers = dict(scope.get("headers", []))
    streaming = b"text/event-stream" in headers.get(b"accept", b"")
    log_start(question, streaming)
    if not streaming:
        try:
            reply, queries, usage = await answer(messages)
        except UnverifiedAnswerError:
            logger.exception("Conversation answer could not be verified")
            await send_response(send, 502, {"error": "unverified_answer"})
            return
        except Exception:
            logger.exception("Conversation upstream failure")
            await send_response(send, 502, {"error": "upstream"})
            return
        logger.info(json.dumps({"event": "conversation", "queries": queries, "usage": usage}))
        await send_response(send, 200, {"reply": reply, "queries": queries})
        return

    await send({
        "type": "http.response.start", "status": 200,
        "headers": [
            (b"content-type", b"text/event-stream"),
            (b"cache-control", b"no-store"),
            (b"x-accel-buffering", b"no"),
        ],
    })

    async def on_text(text: str) -> None:
        # answer() releases text only once it can no longer trip the guard (D-13), so it streams as it comes.
        if text:
            payload = json.dumps({"text": text}).encode()
            await send({
                "type": "http.response.body",
                "body": b"event: delta\ndata: " + payload + b"\n\n", "more_body": True,
            })

    try:
        reply, queries, usage = await answer(messages, on_text=on_text)
    except UnverifiedAnswerError:
        logger.exception("Conversation answer could not be verified")
        await send({
            "type": "http.response.body",
            "body": b'event: error\ndata: {"error":"unverified_answer"}\n\n', "more_body": False,
        })
        return
    except Exception:
        logger.exception("Conversation upstream failure")
        await send({
            "type": "http.response.body",
            "body": b'event: error\ndata: {"error":"upstream"}\n\n', "more_body": False,
        })
        return
    logger.info(json.dumps({"event": "conversation", "queries": queries, "usage": usage}))
    final = json.dumps({"reply": reply, "queries": queries}).encode()
    await send({
        "type": "http.response.body",
        "body": b"event: done\ndata: " + final + b"\n\n", "more_body": False,
    })

def validate_messages(value) -> list[dict]:
    if not isinstance(value, list) or not 1 <= len(value) <= 20:
        raise ValueError("too_long")
    messages = []
    for message in value:
        if not isinstance(message, dict) or message.get("role") not in {"user", "assistant"}:
            raise ValueError("invalid_request")
        content = message.get("content")
        if not isinstance(content, str) or not 1 <= len(content) <= 2000:
            raise ValueError("too_long")
        messages.append({"role": message["role"], "content": content})
    return messages


def result_text(result) -> str:
    parts = [item.text for item in result.content if item.type == "text"]
    if result.structured_content is not None:
        return json.dumps(result.structured_content)
    return "\n".join(parts)


def result_payload(result):
    if result.structured_content is not None:
        return result.structured_content.get("result", result.structured_content)
    return json.loads(result_text(result))


NON_FIGURE_NUMBERS = re.compile(
    r"\b(?:release\s+v?\d+(?:\.\d+)*|question\s+\d+|q-\d+|decision\s+\d+|d-\d+"
    r"|\d{4}-\d{2}-\d{2}|\d{1,2}/\d{1,2}/\d{2,4}"
    r"|(?:january|february|march|april|may|june|july|august|september|october|november|december)"
    r"\s+\d{1,2}(?:,\s*\d{4})?)\b",
    re.IGNORECASE,
)


def states_figure(reply: str) -> bool:
    """Ignore planning labels and dates; every other digit needs query evidence."""
    return any(character.isdigit() for character in NON_FIGURE_NUMBERS.sub("", reply))


def is_throttled(error: BaseException) -> bool:
    if isinstance(error, httpx2.HTTPStatusError):
        return error.response.status_code == 429
    if isinstance(error, BaseExceptionGroup):
        return any(is_throttled(item) for item in error.exceptions)
    return False


async def call_digest(session, name: str, arguments: dict):
    for attempt in range(3):
        try:
            return await session.call_tool(name, arguments)
        except Exception as error:
            if not is_throttled(error) or attempt == 2:
                raise
            await asyncio.sleep(0.5 * (2**attempt))


USAGE_KEYS = (

def cached_tools(tools: list[dict]) -> list[dict]:
    """Keep one stable cache breakpoint after the system prompt and tool list."""
    cached = [dict(tool) for tool in tools]
    if cached:
        cached[-1]["cache_control"] = {"type": "ephemeral"}
    return cached


async def semantic_catalogue(digest) -> list[dict]:
    """Read the current metric catalogue and valid dimensions from MetricFlow."""
    listed = result_payload(await call_digest(digest, "list-metrics", {}))
    dimension_results = []
    for metric in listed:
        dimension_results.append(await call_digest(
            digest, "get-dimensions", {"metric_name": metric["name"]}
        ))
    return [{
        "name": metric["name"],
        "description": metric.get("description", ""),
        "group_by": result_payload(dimensions),
    } for metric, dimensions in zip(listed, dimension_results)]


async def current_semantic_catalogue(digest) -> list[dict]:
    """Reuse the catalogue only while the connector's active manifest is unchanged."""
    global semantic_catalogue_cache, semantic_catalogue_key
    manifest_key = result_payload(await call_digest(digest, "get-manifest-version", {}))
    cache_hit = semantic_catalogue_cache is not None and semantic_catalogue_key == manifest_key
    if not cache_hit:
        catalogue = await semantic_catalogue(digest)
        semantic_catalogue_cache = catalogue
        semantic_catalogue_key = manifest_key
    print(json.dumps({
        "event": "semantic_catalogue", "cache_hit": cache_hit,
        "manifest_version": manifest_key,
    }), flush=True)
    return semantic_catalogue_cache


async def stream_completion(client, payload: dict, headers: dict, on_text) -> dict:
    """Collect one Anthropic SSE response while forwarding its text deltas."""
    content = []
    usage = {}
    async with client.stream(
        "POST", "https://api.anthropic.com/v1/messages", headers=headers,
        json={**payload, "stream": True},
    ) as api_response:
        if api_response.status_code >= 400:
            await api_response.aread()
            logger.error(json.dumps({
                "event": "model_error", "status": api_response.status_code,
                "body": api_response.text,
            }))
        api_response.raise_for_status()
        async for line in api_response.aiter_lines():
            if not line.startswith("data:"):
                continue
            event = json.loads(line[5:].strip())
            kind = event.get("type")
            if kind == "message_start":
                usage.update(event.get("message", {}).get("usage", {}))
            elif kind == "content_block_start":
                block = event["content_block"]
                content.append(dict(block))
                if block.get("type") == "tool_use":
                    content[-1]["_partial_json"] = ""
            elif kind == "content_block_delta":
                index = event["index"]
                delta = event["delta"]
                if delta.get("type") == "text_delta":
                    text = delta.get("text", "")
                    content[index]["text"] = content[index].get("text", "") + text
                    await on_text(text)
                elif delta.get("type") == "thinking_delta":
                    content[index]["thinking"] = (
                        content[index].get("thinking", "") + delta.get("thinking", "")
                    )
                elif delta.get("type") == "signature_delta":
                    content[index]["signature"] = (
                        content[index].get("signature", "") + delta.get("signature", "")
                    )
                elif delta.get("type") == "input_json_delta":
                    content[index]["_partial_json"] += delta.get("partial_json", "")
            elif kind == "message_delta":
                usage.update(event.get("usage", {}))
    for block in content:
        partial = block.pop("_partial_json", None)
        if partial is not None:
            block["input"] = json.loads(partial) if partial else {}
    return {"content": content, "usage": usage}


class UnverifiedAnswerError(Exception):
    """The final reply states numbers but no query-metrics call ran for the question."""


async def answer(messages: list[dict], on_text=None) -> tuple[str, list[dict], dict]:
    model = os.environ.get("ANTHROPIC_MODEL", "claude-sonnet-5-5")
    usage = {key: 0 for key in USAGE_KEYS}
    queries = []
    tool_calls = 0
    tools_enabled = True
    model_call = 0
    number_verification_retry = True
    retry_streams_text = False
    held, released = [], []

    async def gated_text(text):
        # D-13: hold streamed text until the question's first query-metrics call has run; then release
        # what was held and stream the rest as it arrives. Text with no query behind it waits for the guard.
        if queries:
            if held:
                await on_text("".join(held))
                released.append("".join(held))
                held.clear()
            await on_text(text)
            released.append(text)
        else:
            held.append(text)

    async def release_accepted(reply):
        # The guard accepted the reply: release whatever is still held, or the whole reply if nothing streamed.
        if on_text is None:
            return
        if held:
            await on_text("".join(held))
            held.clear()
        elif not released:
            await on_text(reply)

    async with digest_session() as digest:
        listed = await digest.list_tools()
        tools = [{
            "name": tool.name,
            "description": tool.description or "",
            "input_schema": tool.input_schema,
        } for tool in listed.tools if tool.name == "query-metrics"]
        catalogue = await current_semantic_catalogue(digest)
        system_prompt = (
            f"{SYSTEM_PROMPT}\n\nCurrent semantic-layer catalogue:\n"
            f"{json.dumps(catalogue, separators=(',', ':'))}"
        )
        async with httpx2.AsyncClient(timeout=60.0) as client:
            while True:
                model_call += 1
                request_payload = {
                    "model": model,
                    "max_tokens": 1024,
                    "thinking": {"type": "between_tools"},
                    "output_config": {"effort": "low"},
                    "system": [{
                        "type": "text", "text": system_prompt,
                        "cache_control": {"type": "ephemeral"},
                    }],
                    "messages": messages,
                }
                if tools_enabled:
                    request_payload["tools"] = cached_tools(tools)
                headers = {
                    "x-api-key": secret("anthropic-api-key"),
                    "anthropic-version": "2023-06-01",
                }
                started = time.perf_counter()
                stream_text = None if (retry_streams_text or on_text is None) else gated_text
                try:
                    if stream_text is None:
                        api_response = await client.post(
                            "https://api.anthropic.com/v1/messages",
                            headers=headers, json=request_payload,
                        )
                        if api_response.status_code >= 400:
                            logger.error(json.dumps({
                                "event": "model_error", "status": api_response.status_code,
                                "body": api_response.text,
                            }))
                        api_response.raise_for_status()
                        completion = api_response.json()
                    else:
                        completion = await stream_completion(
                            client, request_payload, headers, stream_text
                        )
                except (httpx2.HTTPError, ValueError) as error:
                    raise UpstreamError("Claude request failed") from error
                call_usage = completion.get("usage", {})
                print(json.dumps({
                    "event": "model_call", "environment": environment(), "call": model_call,
                    "duration_ms": round((time.perf_counter() - started) * 1000),
                    "streaming": stream_text is not None, "usage": call_usage,
                }), flush=True)
                for key in usage:
                    usage[key] += call_usage.get(key, 0)
                content = completion.get("content", [])
                messages.append({"role": "assistant", "content": content})
                uses = [block for block in content if block.get("type") == "tool_use"]
                if not uses:
                    reply = "\n".join(
                        block.get("text", "") for block in content if block.get("type") == "text"
                    ).strip()
                    if not reply:
                        raise UpstreamError("Claude returned no answer")
                    if not queries and states_figure(reply):
                        if number_verification_retry:
                            number_verification_retry = False
                            retry_streams_text = True
                            held.clear()  # unverified text is never released
                            messages.append({
                                "role": "user",
                                "content": (
                                    "Your reply states numbers, but no query-metrics call has run "
                                    "for this question in this conversation. Every number in a "
                                    "DV Digest answer must come from a query-metrics call. Run "
                                    "the query-metrics calls you need, then answer the question "
                                    "again."
                                ),
                            })
                            continue
                        raise UnverifiedAnswerError(
                            "Answer could not be verified: it states numbers but no "
                            "query-metrics call ran for this question"
                        )
                    await release_accepted(reply)
                    return reply, queries, usage
                tool_results = []
                execution = []
                for use in uses:
                    name = use.get("name")
                    if name not in {"list-metrics", "get-dimensions", "query-metrics"}:
                        raise UpstreamError("Claude requested an unknown tool")
                    if tool_calls >= 6:
                        tools_enabled = False
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": use["id"],
                            "content": (
                                "Six-tool-call limit reached. Answer from the evidence already "
                                "gathered, or state what is unavailable; do not request more tools."
                            ),
                            "is_error": True,
                        })
                        continue
                    tool_calls += 1
                    execution.append((len(tool_results), use))
                    tool_results.append(None)

                async def execute(use):
                    print(json.dumps({
                        "event": "tool_request", "environment": environment(), "name": use["name"],
                        "arguments": use.get("input") or {},
                    }), flush=True)
                    try:
                        return await call_digest(
                            digest, use["name"], use.get("input") or {}
                        )
                    except Exception as error:
                        raise UpstreamError("DV Digest request failed") from error

                called_results = await asyncio.gather(
                    *(execute(use) for _, use in execution)
                )
                for (index, use), called in zip(execution, called_results):
                    arguments = use.get("input") or {}
                    if use["name"] == "query-metrics" and not called.is_error:
                        try:
                            payload = result_payload(called)
                            rows = len(payload.get("rows", []))
                            labels = payload.get("labels") or {}
                        except (ValueError, TypeError, AttributeError):
                            rows, labels = 0, {}
                        queries.append({
                            "metric": arguments.get("metric_name", ""),
                            "group_by": arguments.get("group_by") or [],
                            "rows": rows,
                            "label": labels.get("metric", ""),
                            "group_by_labels": labels.get("group_by") or [],
                        })
                    tool_results[index] = {
                        "type": "tool_result",
                        "tool_use_id": use["id"],
                        "content": result_text(called),
                        "is_error": called.is_error,
                    }
                messages.append({"role": "user", "content": tool_results})


def handler(event, context):
    early = preflight(event)
    if early is not None:
        return early
    try:
        body = json_body(event)
    except (ValueError, UnicodeDecodeError):
        return response(400, {"error": "invalid_request"})
    if not valid_access_code(body.get("access_code")):
        return response(401, {"error": "access_code"})
    try:
        messages = validate_messages(body.get("messages"))
    except ValueError as error:
        return response(400, {"error": str(error)})
    cap = int(os.environ.get("DIGEST_DAILY_CAP", "200"))
    if not consume_daily_allowance("conversation", cap):
        return response(429, {"error": "daily_cap"})
    question = next(
        (item["content"] for item in reversed(messages) if item["role"] == "user"), ""
    )
    try:
        reply, queries, usage = asyncio.run(answer(messages))
    except UnverifiedAnswerError:
        logger.exception("Conversation answer could not be verified")
        return response(502, {"error": "unverified_answer"})
    except Exception:
        logger.exception("Conversation upstream failure")
        return response(502, {"error": "upstream"})
    logger.info(json.dumps({
        "event": "conversation",
        "environment": environment(),
        "question": question,
        "queries": queries,
        "usage": usage,
    }))
    return response(200, {"reply": reply, "queries": queries})

class LocalManifestSource:
    """Reads the semantic manifest packaged with the dev connector."""

    def __init__(self, path: Path | None = None) -> None:
        self.path = path or Path(__file__).resolve().with_name("dev_semantic_manifest.json")

    def latest_successful_run_id(self) -> int:
        return 1

    def semantic_manifest(self, run_id: int) -> str:
        if run_id != 1:
            raise RuntimeError(f"Unknown local manifest run {run_id}")
        return self.path.read_text(encoding="utf-8")


class DbtCloudManifestSource:
    """Reads the latest successful QA run and its optional semantic manifest."""

    def __init__(
        self,
        token: str,
        host: str,
        account_id: int,
        job_id: int,
        timeout_seconds: int = 30,
    ) -> None:
        self.token = token
        self.host = host.rstrip("/")
        if not self.host.startswith("https://"):
            self.host = f"https://{self.host}"
        self.account_id = account_id
        self.job_id = job_id
        self.timeout_seconds = timeout_seconds

    @classmethod
    def from_environment(cls) -> "DbtCloudManifestSource":
        required = (
            "DBT_CLOUD_TOKEN", "DBT_CLOUD_HOST", "DBT_CLOUD_ACCOUNT_ID", "DBT_CLOUD_JOB_ID"
        )

                or config.get("InvokeMode") != invoke_mode):

            FunctionName=name, AuthType="NONE", InvokeMode=invoke_mode, Cors=cors

def build_package(stage: str) -> tuple[Path, int, int]:

            "--platform", "manylinux2014_x86_64",
            "--python-version", "3.12",
            "--implementation", "cp",
            "--abi", "cp312", "--abi", "abi3", "--abi", "none",
            "--only-binary=:all:",
            "-r", str(CONNECTOR / "requirements-lambda.txt"),

        dirs_exist_ok=True,

            "__pycache__", "*.pyc", "test_*", "deploy_lambda.py"

        )

    startup = package_dir / "run.sh"

    shutil.rmtree(package_dir, ignore_errors=True)

def invite(username: str) -> dict:
    headers = {
        "Authorization": f"Bearer {secret('github-invite-token')}",
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
        "User-Agent": "dig590-dv-digest-invite",
    }
    try:
        with httpx2.Client(timeout=30.0, headers=headers) as client:
            user = client.get(f"https://api.github.com/users/{username}")
            if user.status_code == 404:
                return {"status": 404, "body": {"error": "no_such_user"}}
            user.raise_for_status()
            collaborator_url = (
                "https://api.github.com/repos/rayserrano2735/digest-datavault/"
                f"collaborators/{username}"
            )
            membership = client.get(collaborator_url)
            if membership.status_code == 204:
                return {"status": 200, "body": {"already": True}}
            if membership.status_code != 404:
                membership.raise_for_status()
            cap = int(os.environ.get("DIGEST_INVITE_DAILY_CAP", "5"))
            if not consume_daily_allowance("invite", cap):
                return {"status": 429, "body": {"error": "daily_cap"}}
            sent = False
            try:
                invitation = client.put(collaborator_url, json={"permission": "push"})
                if invitation.status_code in {200, 201}:
                    sent = True
                    return {"status": 200, "body": {"invited": True}}
                if invitation.status_code == 204:
                    return {"status": 200, "body": {"already": True}}
                invitation.raise_for_status()
            finally:
                if not sent:
                    release_daily_allowance("invite")
    except httpx2.HTTPError as error:
        raise RuntimeError("GitHub invitation failed") from error
    raise RuntimeError("GitHub returned an unexpected invitation response")

class SnowflakeRunner:
    """Runs compiled SQL on Snowflake. Opens its session lazily and opens a fresh one when Snowflake says the old
    one can no longer be renewed (error 250001, after the session's master token expires; seen 2026-10-27 after
    the server had run for hours), so a long-running server never needs a restart for it."""

    def __init__(self) -> None:
        private_key = serialization.load_pem_private_key(
            os.environ["SNOWFLAKE_PRIVATE_KEY"].encode(), password=None
        )
        self._connect_args = dict(
            user=os.environ["SNOWFLAKE_USER"],
            account=os.environ["SNOWFLAKE_ACCOUNT"],
            authenticator="SNOWFLAKE_JWT",
            private_key=private_key.private_bytes(
                serialization.Encoding.DER,
                serialization.PrivateFormat.PKCS8,
                serialization.NoEncryption(),
            ),
            role=os.environ["SNOWFLAKE_ROLE"],
            warehouse=os.environ["SNOWFLAKE_WAREHOUSE"],
            client_session_keep_alive=True,
            session_parameters={"TIMEZONE": "UTC"},
        )
        self.connection = None
        self._lock = threading.Lock()

    def _connection(self, fresh: bool = False):
        with self._lock:
            if fresh and self.connection is not None:
                try:
                    self.connection.close()
                except Exception:
                    pass
                self.connection = None
            if self.connection is None or self.connection.is_closed():
                self.connection = snowflake.connector.connect(**self._connect_args)
                # The session runs as its role alone: the user's secondary roles, SYSADMIN among them, would unmask the
                # insured's details and give the connector far more than it reads (QA run 4, 2026-11-01; Ray, answer 1a).
                self.connection.cursor().execute("USE SECONDARY ROLES NONE")
            return self.connection

    def query(self, sql: str) -> list[dict]:
        try:
            return self._run(self._connection(), sql)
        except snowflake.connector.errors.Error as exc:
            # ReauthenticationRequest (250001) on newer connectors; an expired session token (390114) on older ones.
            if type(exc).__name__ != "ReauthenticationRequest" and getattr(exc, "errno", None) not in (250001, 390114):
                raise
            logger.warning("Snowflake session expired (%s); reconnecting once", getattr(exc, "errno", exc))
            return self._run(self._connection(fresh=True), sql)

    @staticmethod
    def _serialize_value(value):
        if isinstance(value, datetime):
            if value.tzinfo is None:
                value = value.replace(tzinfo=timezone.utc)
            else:
                value = value.astimezone(timezone.utc)
        return value.isoformat() if hasattr(value, "isoformat") else value

    @staticmethod
    def _run(connection, sql: str) -> list[dict]:
        with connection.cursor() as cursor:
            cursor.execute(sql)
            columns = [column[0].lower() for column in cursor.description]
            return [
                {
                    column: SnowflakeRunner._serialize_value(value)
                    for column, value in zip(columns, row)
                }
                for row in cursor.fetchall()
            ]


class MetricFlowBackend:
    def __init__(
        self,
        manifest_source=None,
        poll_interval_seconds: float | None = None,
    ) -> None:
        self.manifest = None
        self.engine = None
        self.runner = SnowflakeRunner() if os.environ.get("SNOWFLAKE_ACCOUNT") else None

        self._lock = threading.RLock()
        self._stop_polling = threading.Event()
        self._manifest_source = manifest_source or (
            LocalManifestSource()
            if os.environ.get("DIGEST_STAGE") == "dev"
            else DbtCloudManifestSource.from_environment()
        )
        self._active_run_id: int | None = None
        self._checked_run_id: int | None = None
        self._poll_interval_seconds = (
            poll_interval_seconds
            if poll_interval_seconds is not None
            else float(os.environ.get("DBT_CLOUD_POLL_SECONDS", "300"))
        )

        self.refresh_manifest(initial=True)
        if self._poll_interval_seconds > 0:
            thread = threading.Thread(
                target=self._poll_loop,
                name="dbt-cloud-manifest-poll",
                daemon=True,
            )
            thread.start()

    @staticmethod
    def _build_engine(manifest) -> MetricFlowEngine:
        return MetricFlowEngine(
            semantic_manifest_lookup=SemanticManifestLookup(manifest),
            sql_client=CompileOnlySnowflakeClient(),
        )

    @property
    def active_run_id(self) -> int | None:
        return self._active_run_id

    @property
    def checked_run_id(self) -> int | None:
        return self._checked_run_id

    def refresh_manifest(self, initial: bool = False) -> bool:
        """Require a valid manifest at startup; retain the last good manifest during polling."""
        try:
            run_id = self._manifest_source.latest_successful_run_id()
            if run_id is None:
                if initial:
                    raise RuntimeError("No successful dbt Cloud run is available")
                return False
            if run_id == self._checked_run_id:
                return False
            artifact = self._manifest_source.semantic_manifest(run_id)
            if artifact is None:
                if initial:
                    raise RuntimeError(f"dbt Cloud run {run_id} has no semantic manifest")
                with self._lock:
                    self._checked_run_id = run_id
                logger.info("Run %s has no semantic manifest; retaining the current manifest", run_id)
                return False

            candidate = parse_manifest_from_dbt_generated_manifest(artifact)
            if not candidate.metrics:
                if initial:
                    raise RuntimeError(f"dbt Cloud run {run_id} contains no metrics")
                with self._lock:
                    self._checked_run_id = run_id
                logger.info("Run %s contains no metrics; retaining the current manifest", run_id)
                return False
            candidate_engine = self._build_engine(candidate)
            with self._lock:
                self.manifest = candidate
                self.engine = candidate_engine
                self._active_run_id = run_id
                self._checked_run_id = run_id
            logger.info("Loaded semantic manifest from dbt Cloud run %s", run_id)
            return True
        except Exception:
            if initial:
                logger.error("Unable to load the required initial semantic manifest", exc_info=True)
                raise
            logger.warning("Unable to refresh the semantic manifest; retaining the last good one", exc_info=True)
            return False

    def _poll_loop(self) -> None:
        while not self._stop_polling.wait(self._poll_interval_seconds):
            self.refresh_manifest()

    def list_metrics(self) -> list[dict]:
        with self._lock:
            return [
                {"name": metric.name, "description": metric.description or ""}
                for metric in self.manifest.metrics
            ]

    def get_dimensions(self, metric_name: str) -> list[str]:
        with self._lock:
            if metric_name not in {metric.name for metric in self.manifest.metrics}:
                raise ValueError(f"Unknown governed metric: {metric_name}")
            return sorted(
                dimension.dunder_name
                for dimension in self.engine.list_dimensions(metric_names=[metric_name])
            )

    def compile_query(self, metric_name: str, group_by: list[str]) -> str:
        with self._lock:
            self.get_dimensions(metric_name)
            request = MetricFlowQueryRequest.create(
                metric_names=[metric_name],
                group_by_names=group_by,
            )
            return self.engine.explain(request).sql_statement.sql

    def labels(self, metric_name: str, group_by: list[str]) -> dict:
        """The business labels of a query's metric and groupings, so the consumer is shown meaning, never the layer's
        names (Data Consumption Design, k17ki72, section 11). A grouping such as policy__carrier__carrier_name or
        client__first_known_date__day takes the label of its dimension, carrier_name or first_known_date."""
        with self._lock:
            metric_labels = {m.name: (m.label or "") for m in self.manifest.metrics}
            dimension_labels = {}
            for model in self.manifest.semantic_models:
                for dimension in model.dimensions:
                    if dimension.label:
                        dimension_labels.setdefault(dimension.name, dimension.label)
        def grouping(name: str) -> str:
            parts = name.split("__")
            if len(parts) > 1 and parts[-1] in ("day", "week", "month", "quarter", "year"):
                parts = parts[:-1]
            return dimension_labels.get(parts[-1], "")
        return {"metric": metric_labels.get(metric_name, ""), "group_by": [grouping(g) for g in group_by]}

    def query(self, metric_name: str, group_by: list[str]) -> dict:
        sql = self.compile_query(metric_name, group_by)
        if self.runner is None:
            return {
                "metric_name": metric_name,
                "group_by": group_by,
                "compiled_sql": sql,
                "execution": "compile only; no Snowflake connection configured",
            }
        return {
            "metric_name": metric_name,
            "group_by": group_by,
            "rows": self.runner.query(sql),
            "labels": self.labels(metric_name, group_by),
            "execution": "snowflake",
        }

def response(status: int, body: dict | None = None) -> dict:
    return {
        "statusCode": status,
        # Function URL CORS adds the environment origin exactly once to every reply.
        "headers": {
            "Content-Type": "application/json",
            "Cache-Control": "no-store",
        },
        "body": json.dumps(body or {}),
    }


def preflight(event: dict) -> dict | None:
    method = event.get("requestContext", {}).get("http", {}).get("method")
    if method != "OPTIONS":
        return None
    result = response(204)
    result["headers"].update({
        "Access-Control-Allow-Methods": "POST, OPTIONS",
        "Access-Control-Allow-Headers": "Content-Type",
        "Access-Control-Max-Age": "300",
    })
    result["body"] = ""
    return result


def json_body(event: dict) -> dict:
    body = event.get("body") or "{}"
    if event.get("isBase64Encoded"):
        import base64
        body = base64.b64decode(body).decode("utf-8")
    value = json.loads(body)
    if not isinstance(value, dict):
        raise ValueError("JSON body must be an object")
    return value


def secret(suffix: str) -> str:
    if suffix not in _secret_cache:
        prefix = os.environ["SSM_PARAMETER_PREFIX"]
        value = boto3.client("ssm", region_name=os.environ.get("AWS_REGION", "us-east-1"))
        parameter = value.get_parameter(Name=f"{prefix}/{suffix}", WithDecryption=True)
        _secret_cache[suffix] = parameter["Parameter"]["Value"]
    return _secret_cache[suffix]


def valid_access_code(provided) -> bool:
    if not isinstance(provided, str) or not provided:
        return False
    return hmac.compare_digest(provided.encode("utf-8"), secret("access-code").encode("utf-8"))


def consume_daily_allowance(kind: str, cap: int) -> bool:
    """Increment a small per-environment UTC-day counter in Standard Parameter Store."""
    day = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    prefix = os.environ["SSM_PARAMETER_PREFIX"]
    name = f"{prefix}/daily-{kind}-{day}"
    client = boto3.client("ssm", region_name=os.environ.get("AWS_REGION", "us-east-1"))
    try:
        current = int(client.get_parameter(Name=name)["Parameter"]["Value"])
    except ClientError as error:
        if error.response["Error"]["Code"] != "ParameterNotFound":
            raise
        current = 0
    if current >= cap:
        return False
    client.put_parameter(Name=name, Value=str(current + 1), Type="String", Overwrite=True)
    return True


def release_daily_allowance(kind: str) -> None:
    """Release a reserved invitation when GitHub did not send one."""
    day = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    prefix = os.environ["SSM_PARAMETER_PREFIX"]
    name = f"{prefix}/daily-{kind}-{day}"
    client = boto3.client("ssm", region_name=os.environ.get("AWS_REGION", "us-east-1"))
    try:
        current = int(client.get_parameter(Name=name)["Parameter"]["Value"])
    except ClientError as error:
        if error.response["Error"]["Code"] == "ParameterNotFound":
            return
        raise
    client.put_parameter(
        Name=name, Value=str(max(0, current - 1)), Type="String", Overwrite=True
    )


class LambdaSigV4Auth(httpx2.Auth):
    """Sign Function URL requests without exposing credentials to application code."""

    def __init__(self):
        credentials = boto3.Session(
            region_name=os.environ.get("AWS_REGION", "us-east-1")
        ).get_credentials()
        if credentials is None:
            raise RuntimeError("AWS credentials are unavailable")
        self.credentials = credentials.get_frozen_credentials()

    def auth_flow(self, request):
        aws_request = AWSRequest(
            method=request.method,
            url=str(request.url),
            headers=dict(request.headers),
            data=request.content,
        )
        SigV4Auth(
            self.credentials,
            "lambda",
            os.environ.get("AWS_REGION", "us-east-1"),
        ).add_auth(aws_request)
        request.headers.update(dict(aws_request.headers))
        yield request

"""
The vault's master DDL (Ray, 2026-10-28, answers 2y and 3y).

Phase 1: dbt creates the structures, so the DDL is reverse-engineered from a built vault with Snowflake's GET_DDL,
for the schemas a modeler works with: RAW_VAULT, REFERENCE and CONFORMED_DM (MART retired, mart design D-6). Database names are written as production's
(DV, LANDING), so the one file describes every tier and can be compared with any of them.

Generate, on a release's branch, right after a dbt build in the developer's database:
  python3 infra/scripts/vault_ddl.py --database DEV__DV__CLAUDE_WS --landing DEV__LANDING__CLAUDE_WS -- <snow options>
Check a tier's deployed structure against the committed file (QA after its build, production after its build):
  python3 infra/scripts/vault_ddl.py --check --database DV_QA --landing LANDING_QA -- <snow options>
  python3 infra/scripts/vault_ddl.py --check --database DV --landing LANDING -- <snow options>
<snow options> are the Snowflake CLI's connection options, as the workstation's other scripts pass them.
--check exits 1 and shows the differences when the structure is not the one on record.
"""
import argparse, difflib, json, os, re, subprocess, sys

SCHEMAS = ["RAW_VAULT", "REFERENCE", "CONFORMED_DM"]
OUTPUT = os.path.join(os.path.dirname(__file__), "..", "snowflake", "ddl", "dv.sql")
HEADER = """-- The vault's structure: the raw vault, reference and conformed mart schemas of the DIGEST Data Vault course.
-- Phase 1: generated from a built vault by infra/scripts/vault_ddl.py (Snowflake GET_DDL), with database names
-- written as production's (DV, LANDING). Each release commits its own version.
-- To propose a change to the structure, edit this file on a branch of your own: the diff is the proposal.
"""


def ddl(database, landing, snow):
    """The normalized DDL of the three schemas, in a fixed order."""
    sql = "\n".join(f"SELECT '{s}' AS S, GET_DDL('SCHEMA', '{database}.{s}', FALSE) AS D;" for s in SCHEMAS)
    run = subprocess.run(["snow", "sql", "-i", "--format", "json", "--enable-templating", "NONE"] + snow,
                         input=sql.encode(), capture_output=True)
    out = run.stdout.decode("utf-8", "replace")
    try:
        results = json.loads(out)
    except ValueError:
        sys.exit("GET_DDL failed:\n" + out[:2000] + run.stderr.decode("utf-8", "replace")[:1000])
    parts = []
    for rows in results:
        text = rows[0]["D"]
        for name, prod in ((database, "DV"), (landing, "LANDING")):
            text = re.sub(rf"\b{re.escape(name)}\b", prod, text, flags=re.IGNORECASE)
        # dbt signs each view's text with a comment naming its engine, version, target and run; it is not structure, and
        # it differs between Fusion and dbt-core and on every run, so it is removed (QA run 2, 2026-11-01)
        text = re.sub(r'/\* \{"app": "dbt".*?\} \*/', "", text)
        parts.append("\n".join(line.rstrip() for line in text.strip().splitlines()))
    return HEADER + "\n" + "\n\n".join(parts) + "\n"


def main():
    if "--" not in sys.argv:
        sys.exit("usage: vault_ddl.py [--check] --database DB --landing DB -- <snow connection options>")
    i = sys.argv.index("--")
    ap = argparse.ArgumentParser()
    ap.add_argument("--database", required=True)
    ap.add_argument("--landing", required=True)
    ap.add_argument("--check", action="store_true")
    args = ap.parse_args(sys.argv[1:i])
    text = ddl(args.database.upper(), args.landing.upper(), sys.argv[i + 1:])
    if args.check:
        with open(OUTPUT, encoding="utf-8") as f:
            recorded = f.read()
        if text == recorded:
            print(f"{args.database}: structure matches infra/snowflake/ddl/dv.sql")
            return
        diff = list(difflib.unified_diff(recorded.splitlines(), text.splitlines(), "dv.sql (committed)",
                                         f"{args.database} (deployed)", lineterm="", n=1))
        print(f"{args.database}: structure DIFFERS from infra/snowflake/ddl/dv.sql ({len(diff)} diff lines)")
        print("\n".join(diff[:80]))
        sys.exit(1)
    os.makedirs(os.path.dirname(OUTPUT), exist_ok=True)
    with open(OUTPUT, "w", encoding="utf-8") as f:
        f.write(text)
    print(f"wrote infra/snowflake/ddl/dv.sql from {args.database}: {text.count(chr(10))} lines, "
          f"{len(re.findall(r'create or replace (?:transient )?table', text, re.I))} tables, "
          f"{len(re.findall(r'create or replace view', text, re.I))} views")


if __name__ == "__main__":
    main()

/*
  Refresh one non-production landing tier from production landing (Ray, 2026-10-26, answer 1y; generalized to every
  tier 2026-10-28). The tiers are environment_tiers.sql's: LANDING_QA built by DV_QA_ROLE, and each developer's
  DEV__LANDING__<USER> built by DEV_<USER>_ROLE. The tier is replaced by a fresh clone of LANDING, owned by
  LANDING_ROLE, production's builder (DV_ROLE) removed, the tier's own builder and the reader granted, and its CONFIG
  emptied: production's builds write their findings and baselines into LANDING.CONFIG and the clone copies them
  (found 2026-10-28: 3 findings and 34 baselines), so every tier starts with no findings and no baselines of its own.

  Run by the workstation as SYSADMIN, statements on standard input, naming the tier:
    QA:          snow sql ... -D landing=LANDING_QA -D builder=DV_QA_ROLE < infra/snowflake/refresh_landing_tier.sql
    Development: snow sql ... -D landing=DEV__LANDING__CLAUDE_WS -D builder=DEV_CLAUDE_WS_ROLE < (the same)
  Safe to rerun: the clone is replaced each time. Refuses production: the target must not be LANDING.
*/
USE ROLE SYSADMIN;
USE WAREHOUSE COMPUTE_WH;

/* 0. Refuse production and unknown tiers before anything is replaced. */
EXECUTE IMMEDIATE $$
BEGIN
  IF ('<% landing %>' NOT IN ('LANDING_QA') AND NOT STARTSWITH('<% landing %>', 'DEV__LANDING__')) THEN
    RETURN 1/0;  -- stops the script: only LANDING_QA or a developer's DEV__LANDING__<USER> may be refreshed
  END IF;
  RETURN 'refreshing <% landing %> for <% builder %>';
END;
$$;

/* 1. A fresh clone of production landing. */
CREATE OR REPLACE DATABASE <% landing %> CLONE LANDING;

/* 2. Owned by LANDING_ROLE, grants preserved. */
GRANT OWNERSHIP ON DATABASE <% landing %> TO ROLE LANDING_ROLE COPY CURRENT GRANTS;
GRANT OWNERSHIP ON ALL SCHEMAS IN DATABASE <% landing %> TO ROLE LANDING_ROLE COPY CURRENT GRANTS;
GRANT OWNERSHIP ON ALL TABLES  IN DATABASE <% landing %> TO ROLE LANDING_ROLE COPY CURRENT GRANTS;

/* 3. Builder and reader grants, as the owner. Database-level grants do not survive a clone, but the clone's tables
   keep production's object grants, so production's builder, DV_ROLE, is removed first: each tier is built by its
   own builder alone (Ray, 2026-10-26, answer 1b). */
USE ROLE LANDING_ROLE;
REVOKE ALL PRIVILEGES ON ALL TABLES IN DATABASE <% landing %> FROM ROLE DV_ROLE;
REVOKE ALL PRIVILEGES ON ALL SCHEMAS IN DATABASE <% landing %> FROM ROLE DV_ROLE;
REVOKE ALL PRIVILEGES ON DATABASE <% landing %> FROM ROLE DV_ROLE;
GRANT USAGE ON DATABASE <% landing %> TO ROLE <% builder %>;
GRANT USAGE ON ALL SCHEMAS IN DATABASE <% landing %> TO ROLE <% builder %>;
GRANT SELECT ON ALL TABLES IN DATABASE <% landing %> TO ROLE <% builder %>;
GRANT CREATE TABLE, CREATE VIEW, MODIFY ON SCHEMA <% landing %>.STORAGE TO ROLE <% builder %>;
GRANT CREATE TABLE, CREATE VIEW, MODIFY ON SCHEMA <% landing %>.CONFIG TO ROLE <% builder %>;
GRANT INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA <% landing %>.CONFIG TO ROLE <% builder %>;
GRANT USAGE ON DATABASE <% landing %> TO ROLE LANDING_READ_ROLE;
GRANT USAGE ON ALL SCHEMAS IN DATABASE <% landing %> TO ROLE LANDING_READ_ROLE;
GRANT SELECT ON ALL TABLES IN DATABASE <% landing %> TO ROLE LANDING_READ_ROLE;

/* 4. The tier's own findings and baselines start empty: the clone copied production's. As the owner. */
TRUNCATE TABLE <% landing %>.CONFIG.DATA_FINDING;
TRUNCATE TABLE <% landing %>.CONFIG.BUILD_BASELINE;
USE ROLE SYSADMIN;

/* 5. Check: every LANDING source table has its twin in the tier with the same row count; CONFIG is checked in step
   6 instead, since the tier's is emptied on purpose. Returns zero rows when equal. */
SELECT p.TABLE_SCHEMA, p.TABLE_NAME, p.ROW_COUNT AS LANDING_ROWS, q.ROW_COUNT AS TIER_ROWS
FROM LANDING.INFORMATION_SCHEMA.TABLES p
FULL JOIN <% landing %>.INFORMATION_SCHEMA.TABLES q
  ON q.TABLE_SCHEMA = p.TABLE_SCHEMA AND q.TABLE_NAME = p.TABLE_NAME
WHERE COALESCE(p.TABLE_SCHEMA, q.TABLE_SCHEMA) NOT IN ('INFORMATION_SCHEMA', 'CONFIG')
  AND (p.TABLE_NAME IS NULL OR q.TABLE_NAME IS NULL OR p.ROW_COUNT <> q.ROW_COUNT);

/* 6. Check: the tier's CONFIG is empty. Returns 0 and 0. */
SELECT (SELECT COUNT(*) FROM <% landing %>.CONFIG.DATA_FINDING) AS TIER_FINDINGS,
       (SELECT COUNT(*) FROM <% landing %>.CONFIG.BUILD_BASELINE) AS TIER_BASELINES;
