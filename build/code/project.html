dbt/macros/declare_vault_keys.sql
{#
  The vault's declared keys, from one definition: its layer 1 tests (Ray, 2026-10-28, answers 2y and 4y).
  Each unique test on one column, or dbt_utils.unique_combination_of_columns, becomes its table's primary key;
  each relationships test becomes a foreign key to the table it names. Snowflake keeps them as information, not
  enforced, so loads never fail on them; GET_DDL then reports them, and infra/snowflake/ddl/vault.sql and mart.sql carry them
  for a modeling tool. Only tables and seeds are keyed (views cannot be), and a key already present is left alone,
  so this is safe to run after every build. Runs as the project's on-run-end hook.
#}
{% macro declare_vault_keys() %}
  {% if not execute %}{{ return('') }}{% endif %}
  {% set keyed = ['table', 'incremental', 'seed'] %}
  {% set by_name = {} %}
  {% for n in graph.nodes.values() %}
    {% if n.resource_type in ['model', 'seed'] %}{% do by_name.update({n.name: n}) %}{% endif %}
  {% endfor %}
  {% set pks = {} %}
  {% set fks = [] %}
  {% for t in graph.nodes.values() %}
    {% if t.resource_type == 'test' and t.test_metadata and t.attached_node and t.attached_node in graph.nodes %}
      {% set owner = graph.nodes[t.attached_node] %}
      {% if owner.config.materialized in keyed %}
        {% set kw = t.test_metadata.kwargs %}
        {% if t.test_metadata.name == 'unique' and kw.column_name %}
          {% do pks.update({owner.name: [kw.column_name]}) %}
        {% elif t.test_metadata.name == 'unique_combination_of_columns' %}
          {% do pks.update({owner.name: kw.combination_of_columns}) %}
        {% elif t.test_metadata.name == 'relationships' %}
          {% set target = (kw.to | string).split("'")[1] if "'" in (kw.to | string) else (kw.to | string) %}
          {% do fks.append({'table': owner.name, 'column': kw.column_name, 'to': target, 'field': kw.field}) %}
        {% endif %}
      {% endif %}
    {% endif %}
  {% endfor %}

  {% set statements = [] %}
  {% for table, cols in pks.items() | sort %}
    {% set n = by_name[table] %}
    {% do statements.append({'table': n, 'name': 'pk_' ~ table,
        'sql': 'primary key (' ~ cols | join(', ') ~ ')'}) %}
  {% endfor %}
  {% for fk in fks | sort(attribute='table') %}
    {% set n = by_name[fk.table] %}
    {% set target = by_name.get(fk.to) %}
    {% if target and target.config.materialized in keyed and fk.to in pks %}
      {% do statements.append({'table': n, 'name': 'fk_' ~ fk.table ~ '_' ~ fk.column,
          'sql': 'foreign key (' ~ fk.column ~ ') references ' ~ target.database ~ '.' ~ target.schema ~ '.' ~ target.alias ~ ' (' ~ fk.field ~ ')'}) %}
    {% endif %}
  {% endfor %}

  {% set added = [] %}
  {% for s in statements %}
    {% set rel = s.table.database ~ '.' ~ s.table.schema ~ '.' ~ s.table.alias %}
    {% set present = run_query("select count(*) from " ~ s.table.database ~ ".information_schema.table_constraints where table_schema = upper('" ~ s.table.schema ~ "') and table_name = upper('" ~ s.table.alias ~ "') and constraint_name = upper('" ~ s.name ~ "')") %}
    {% if present.columns[0].values()[0] == 0 %}
      {% do run_query('alter table ' ~ rel ~ ' add constraint ' ~ s.name ~ ' ' ~ s.sql) %}
      {% do added.append(s.name) %}
    {% endif %}
  {% endfor %}
  {{ log('declare_vault_keys: ' ~ statements | length ~ ' keys defined, ' ~ added | length ~ ' added', info=True) }}
{% endmacro %}
dbt/macros/development/dev_drop_orphans.sql
{#
  Development cleanup (Ray, 2026-10-23): a build never drops what the project no longer defines, so removed
  or renamed models leave objects behind. This macro lists every table and view in the target database that
  no model, seed or snapshot defines, and drops them only when dry_run is false. It refuses any database whose
  name does not start with DEV__: cleanup is legitimate only in developer databases.
    List:  dbt run-operation dev_drop_orphans
    Drop:  dbt run-operation dev_drop_orphans --args '{dry_run: false}'
  Schemas left empty by the drop, and not used by the project, are dropped as well.
#}
{% macro dev_drop_orphans(dry_run=true) %}
  {%- set db = target.database | upper -%}
  {%- if not db.startswith('DEV__') -%}
    {{ exceptions.raise_compiler_error("Refused: " ~ db ~ " is not a developer database (DEV__...). Cleanup never runs in QA or production.") }}
  {%- endif -%}
  {%- set defined = [] -%}
  {%- set project_schemas = [] -%}
  {%- for node in graph.nodes.values() if node.resource_type in ['model', 'seed', 'snapshot'] and (node.database | upper) == db -%}
    {%- do defined.append((node.schema | upper) ~ '.' ~ ((node.alias or node.name) | upper)) -%}
    {%- if (node.schema | upper) not in project_schemas %}{% do project_schemas.append(node.schema | upper) %}{% endif -%}
  {%- endfor -%}
  {%- set existing = run_query("SELECT table_schema, table_name, table_type FROM " ~ db ~ ".information_schema.tables WHERE table_schema NOT IN ('INFORMATION_SCHEMA', 'PUBLIC') ORDER BY 1, 2") -%}
  {%- set orphans = [] -%}
  {%- for row in existing.rows -%}
    {%- if (row[0] ~ '.' ~ row[1]) not in defined -%}{%- do orphans.append(row) -%}{%- endif -%}
  {%- endfor -%}
  {%- set schemas = run_query("SELECT schema_name FROM " ~ db ~ ".information_schema.schemata WHERE schema_name NOT IN ('INFORMATION_SCHEMA', 'PUBLIC')") -%}
  {%- set orphan_schemas = [] -%}
  {%- for s in schemas.rows if s[0] not in project_schemas -%}{%- do orphan_schemas.append(s[0]) -%}{%- endfor -%}
  {{ log(db ~ ": " ~ (orphans | length) ~ " objects and " ~ (orphan_schemas | length) ~ " schemas not defined by the project" ~ (" (dry run, nothing dropped)" if dry_run else ""), info=true) }}
  {%- for row in orphans -%}
    {{ log("  " ~ row[2] ~ " " ~ row[0] ~ "." ~ row[1], info=true) }}
    {%- if not dry_run -%}
      {%- do run_query("DROP " ~ ("VIEW" if row[2] == 'VIEW' else "TABLE") ~ " IF EXISTS " ~ db ~ "." ~ row[0] ~ "." ~ row[1]) -%}
    {%- endif -%}
  {%- endfor -%}
  {%- for s in orphan_schemas -%}
    {{ log("  SCHEMA " ~ s, info=true) }}
    {%- if not dry_run -%}{%- do run_query("DROP SCHEMA IF EXISTS " ~ db ~ "." ~ s) -%}{%- endif -%}
  {%- endfor -%}
{% endmacro %}

dbt/macros/materializations/merge_into.sql
-- Custom materialization: merge_into
-- Bridges dbt's one-model-one-table constraint to UDM's multi-source integration.
-- Each source model writes independently to a shared DDL-created target table.
-- The model's SELECT is the transformation. The MERGE is the load. No buffer.
--
-- Config (inside meta):
--   target_database: 'LANDING'      (optional, defaults to current database)
--   target_schema: 'INTEGRATION'
--   target_table: 'PARTY'
--   unique_key: 'PARTY_HKEY'
--   diff_key: 'PARTY_HDIFF'         (optional, for change detection)
--   ts_key: 'LOAD_TS_UTC'           (optional, for newer-wins resolution)
--   exclude_from_update: ['CREATED_AT','CREATED_BY']  (optional)
--   retraction_where: 'tgt.CATEGORY_ID IN (...)'       (optional)
--     Ownership scope for retraction. After the MERGE, rows matching this
--     predicate (alias tgt) that are ABSENT from the model's current emission
--     are DELETED. Cures the sparse-scheme orphan: a model that emits rows
--     only for members of a computed set otherwise
--     leaves stale rows behind when a member drops out of the set, because
--     MERGE only touches rows present in the source. Total schemes self-heal
--     and need no scope; sparse schemes declare the partition they own.
--
-- Author: Ray Serrano, March 2026

{% materialization merge_into, adapter='snowflake' %}

    {%- set meta = config.get('meta', {}) -%}
    {%- set target_database = meta.get('target_database', target.database) -%}
    {%- set target_schema = meta.get('target_schema', target.schema) -%}
    {%- set target_table = meta.get('target_table') -%}
    {%- set unique_key = meta.get('unique_key') -%}
    {%- set diff_key = meta.get('diff_key') -%}
    {%- set ts_key = meta.get('ts_key', 'LOAD_TS_UTC') -%}
    {%- set exclude_from_update = meta.get('exclude_from_update', []) -%}
    {%- set retraction_where = meta.get('retraction_where') -%}
    {#- DAT-459 (Ray, 2026-09-25): a cleanup condition names its tables as tokens, <<ref:model_name>> or
        <<source:source_name.table_name>>, and they are resolved here, at run time. A ref() or source() written in the
        model's meta is rendered at parse time to a stand-in name (the calling model's schema and the referenced model's
        name), which is wrong whenever the referenced object lives elsewhere. The model declares each token's dependency
        with a depends_on comment. -#}
    {%- if retraction_where -%}
        {%- set ns = namespace(w=retraction_where) -%}
        {%- for kind, name in modules.re.findall('<<(ref|source):([A-Za-z0-9_.]+)>>', retraction_where) -%}
            {%- if kind == 'ref' -%}
                {%- set rel = ref(name) -%}
            {%- else -%}
                {%- set parts = name.split('.') -%}
                {%- set rel = source(parts[0], parts[1]) -%}
            {%- endif -%}
            {%- set ns.w = ns.w.replace('<<' ~ kind ~ ':' ~ name ~ '>>', rel | string) -%}
        {%- endfor -%}
        {%- set retraction_where = ns.w -%}
    {%- endif -%}

    {% if not target_table %}
        {{ exceptions.raise_compiler_error("merge_into requires meta.target_table") }}
    {% endif %}
    {% if not unique_key %}
        {{ exceptions.raise_compiler_error("merge_into requires meta.unique_key") }}
    {% endif %}

    -- Build the target relation (DDL-created table, not dbt-managed)
    {%- set target_relation = api.Relation.create(
        database=target_database,
        schema=target_schema,
        identifier=target_table
    ) -%}

    -- Compile the model SQL (the SELECT that produces source rows)
    {%- set source_sql = sql -%}

    -- Get column names from the target table for UPDATE SET and INSERT
    {%- set columns = adapter.get_columns_in_relation(target_relation) -%}
    {%- set column_names = columns | map(attribute='name') | list -%}
    {%- set update_columns = [] -%}
    {%- for col in column_names -%}
        {%- if col != unique_key and col not in exclude_from_update -%}
            {%- do update_columns.append(col) -%}
        {%- endif -%}
    {%- endfor -%}

    -- Build MERGE statement
    {%- set merge_sql -%}
        MERGE INTO {{ target_relation }} AS tgt
        USING (
            {{ source_sql }}
        ) AS src
        ON tgt.{{ unique_key }} = src.{{ unique_key }}

        {% if diff_key %}
        WHEN MATCHED
            AND tgt.{{ diff_key }} != src.{{ diff_key }}
            {% if ts_key %}
            AND src.{{ ts_key }} >= tgt.{{ ts_key }}
            {% endif %}
        THEN UPDATE SET
            {% for col in update_columns %}
            tgt.{{ col }} = src.{{ col }}{% if not loop.last %},{% endif %}
            {% endfor %}
        {% else %}
        WHEN MATCHED THEN UPDATE SET
            {% for col in update_columns %}
            tgt.{{ col }} = src.{{ col }}{% if not loop.last %},{% endif %}
            {% endfor %}
        {% endif %}

        WHEN NOT MATCHED THEN INSERT (
            {% for col in column_names %}
            {{ col }}{% if not loop.last %},{% endif %}
            {% endfor %}
        )
        VALUES (
            {% for col in column_names %}
            src.{{ col }}{% if not loop.last %},{% endif %}
            {% endfor %}
        )
    {%- endset -%}

    {{ log("MERGE INTO " ~ target_relation ~ " using model " ~ model.unique_id, info=True) }}

    {%- call statement('main') -%}
        {{ merge_sql }}
    {%- endcall -%}

    {% if retraction_where %}
    -- Retraction: delete rows in this model's ownership scope that the
    -- current emission no longer contains. NOT IN is safe here because unique
    -- keys are generated surrogate hashes and never NULL.
    {%- set retraction_sql -%}
        DELETE FROM {{ target_relation }} tgt
        WHERE ({{ retraction_where }})
          AND tgt.{{ unique_key }} NOT IN (
              SELECT src.{{ unique_key }} FROM (
                  {{ source_sql }}
              ) src
          )
    {%- endset -%}

    {{ log("RETRACTION on " ~ target_relation ~ " for model " ~ model.unique_id, info=True) }}

    {%- call statement('retraction') -%}
        {{ retraction_sql }}
    {%- endcall -%}
    {% endif %}

    {{ return({'relations': []}) }}

{% endmaterialization %}

dbt/macros/verification/baseline_metrics.sql
{%- macro baseline_metrics(db, ls) -%}
{#
  The measured set shared by capture_baseline and verify_build (guide decision G-4; increment 2, guide step 32):
  every landed source table, every vault table by row count, and the findings by finder type. A vault table not
  yet built counts 0, so a first build's 'before' snapshot does not fail.
#}
{%- set vault_tables = ['CLIENT_H', 'POLICY_H', 'INSURED_H', 'AGENT_H', 'CLIENT_POLICY_L', 'INSURED_POLICY_L', 'AGENT_POLICY_L',
                        'CLIENT_SUNBELT_S', 'INSURED_NORTHFIELD_S', 'POLICY_NORTHFIELD_S', 'CLIENT_POLICY_SUNBELT_S'] -%}
        SELECT 'SUNBELT.CLIENT' AS table_name, 'row_count' AS metric_name, COUNT(*) AS metric_value FROM {{ ls }}.SUNBELT.CLIENT
        UNION ALL SELECT 'SUNBELT.POLICY_SALE', 'row_count', COUNT(*) FROM {{ ls }}.SUNBELT.POLICY_SALE
        UNION ALL SELECT 'NORTHFIELD.POLICY', 'row_count', COUNT(*) FROM {{ ls }}.NORTHFIELD.POLICY
{%- for t in vault_tables %}
{%- set rel = adapter.get_relation(database=db, schema='RAW_VAULT', identifier=t) %}
        UNION ALL SELECT '{{ t }}', 'row_count', ({{ "SELECT COUNT(*) FROM " ~ db ~ ".RAW_VAULT." ~ t if rel is not none else "SELECT 0" }})
{%- endfor %}
        UNION ALL SELECT 'DATA_FINDING', 'row_count', COUNT(*) FROM {{ ls }}.CONFIG.DATA_FINDING
        UNION ALL SELECT 'DATA_FINDING', 'row_count|inspection', COALESCE(COUNT_IF(finder_type = 'inspection'), 0) FROM {{ ls }}.CONFIG.DATA_FINDING
        UNION ALL SELECT 'DATA_FINDING', 'row_count|audit', COALESCE(COUNT_IF(finder_type = 'audit'), 0) FROM {{ ls }}.CONFIG.DATA_FINDING
{%- endmacro -%}

dbt/macros/verification/verify_build.sql
{%- macro verify_build() -%}

{#
  Captures the 'after' snapshot, compares it with capture_baseline's 'before', then
  evaluates the declared expectations in seed_build_expectations. Run last:
    dbt run-operation verify_build
  WARN if a guardrail is violated (a table loses more than half its rows) or an
  expectation is not met; LOOKOUT only when everything passes.
  The measured set is this vault's (guide decision G-4); the expectations seed is read from REFERENCE, where
  dbt_project.yml places seeds (2026-10-23; supersedes guide decision G-3).
#}

{% set db = target.database %}
{% set ls = landing_storage() %}

{% do run_query("
    DELETE FROM " ~ ls ~ ".CONFIG.BUILD_BASELINE
    WHERE snapshot_type = 'after' AND database_name = '" ~ db ~ "'
") %}

{#- The measured set lives in baseline_metrics (increment 2, guide step 32). -#}
{% do run_query("
    INSERT INTO " ~ ls ~ ".CONFIG.BUILD_BASELINE (snapshot_type, table_name, metric_name, metric_value, database_name)
    SELECT 'after', table_name, metric_name, metric_value, '" ~ db ~ "'
    FROM (" ~ baseline_metrics(db, ls) ~ ")
") %}

{# ================================================================ #}
{# COMPARE BEFORE VS AFTER                                         #}
{# ================================================================ #}

{% set comparison = run_query("
    WITH before_snap AS (
        SELECT table_name, metric_name, metric_value
        FROM " ~ ls ~ ".CONFIG.BUILD_BASELINE
        WHERE snapshot_type = 'before' AND database_name = '" ~ db ~ "'
    ),
    after_snap AS (
        SELECT table_name, metric_name, metric_value
        FROM " ~ ls ~ ".CONFIG.BUILD_BASELINE
        WHERE snapshot_type = 'after' AND database_name = '" ~ db ~ "'
    )
    SELECT
        COALESCE(b.table_name, a.table_name) AS table_name,
        COALESCE(b.metric_name, a.metric_name) AS metric_name,
        b.metric_value AS before_value,
        a.metric_value AS after_value,
        COALESCE(a.metric_value, 0) - COALESCE(b.metric_value, 0) AS delta,
        CASE
            WHEN b.metric_value IS NULL THEN 'NEW'
            WHEN a.metric_value IS NULL THEN 'GONE'
            WHEN a.metric_value = b.metric_value THEN 'UNCHANGED'
            WHEN a.metric_value > b.metric_value THEN 'INCREASED'
            WHEN a.metric_value < b.metric_value THEN 'DECREASED'
        END AS status,
        CASE
            WHEN b.metric_value IS NOT NULL AND b.metric_value > 0
                AND (COALESCE(a.metric_value, 0) / b.metric_value) < 0.5
            THEN 'GUARDRAIL'
            ELSE ''
        END AS alert
    FROM before_snap b
    FULL OUTER JOIN after_snap a
        ON b.table_name = a.table_name
        AND b.metric_name = a.metric_name
    ORDER BY COALESCE(b.table_name, a.table_name), COALESCE(b.metric_name, a.metric_name)
") %}

{# ================================================================ #}
{# DISPLAY COMPARISON                                               #}
{# ================================================================ #}

{{ log("", info=True) }}
{{ log("========== BUILD VERIFICATION: " ~ db ~ " ==========", info=True) }}
{{ log("", info=True) }}
{{ log("--- COMPARISON ---", info=True) }}

{% set failures = [] %}

{% for row in comparison %}
    {% set tbl = row[0] %}
    {% set metric = row[1] %}
    {% set before = row[2] %}
    {% set after = row[3] %}
    {% set delta = row[4] %}
    {% set status = row[5] %}
    {% set alert = row[6] | trim %}

    {% set before_str = before | string if before is not none else "—" %}
    {% set after_str = after | string if after is not none else "—" %}
    {% set delta_prefix = "+" if delta > 0 else "" %}

    {% set line = tbl ~ " | " ~ metric ~ " | " ~ before_str ~ " -> " ~ after_str ~ " (" ~ delta_prefix ~ delta | string ~ ") " ~ status %}

    {% if alert == 'GUARDRAIL' %}
        {{ log("  !!! " ~ line ~ " — GUARDRAIL: >50% DROP", info=True) }}
        {% do failures.append("GUARDRAIL: " ~ tbl ~ " " ~ metric ~ " dropped >50%") %}
    {% elif status == 'NEW' %}
        {{ log("  +++ " ~ line, info=True) }}
    {% elif status == 'GONE' %}
        {{ log("  --- " ~ line, info=True) }}
    {% elif status != 'UNCHANGED' %}
        {{ log("  >>> " ~ line, info=True) }}
    {% else %}
        {{ log("      " ~ line, info=True) }}
    {% endif %}
{% endfor %}

{# ================================================================ #}
{# EVALUATE EXPECTATIONS                                            #}
{# ================================================================ #}

{{ log("", info=True) }}
{{ log("--- EXPECTATIONS ---", info=True) }}

{% set expectations = run_query("
    SELECT e.version_tag, e.table_name, e.metric_name, e.condition, e.description,
           CAST(a.metric_value AS FLOAT) AS actual_value
    FROM " ~ db ~ ".REFERENCE.SEED_BUILD_EXPECTATIONS e
    LEFT JOIN " ~ ls ~ ".CONFIG.BUILD_BASELINE a
        ON a.table_name = e.table_name
        AND a.metric_name = e.metric_name
        AND a.snapshot_type = 'after'
        AND a.database_name = '" ~ db ~ "'
    ORDER BY e.version_tag, e.table_name, e.metric_name
") %}

{% for row in expectations %}
    {% set version_tag = row[0] %}
    {% set tbl = row[1] %}
    {% set metric = row[2] %}
    {% set condition = row[3] | trim %}
    {% set description = row[4] %}
    {% set actual = row[5] %}

    {% set passed = false %}

    {% if actual is none %}
        {% set passed = false %}
    {% elif condition == '> 0' %}
        {% set passed = (actual > 0) %}
    {% elif condition.startswith('>= ') %}
        {% set threshold = condition.replace('>= ', '') | float %}
        {% set passed = (actual >= threshold) %}
    {% elif condition.startswith('> ') %}
        {% set threshold = condition.replace('> ', '') | float %}
        {% set passed = (actual > threshold) %}
    {# a declared decrease (Ray, 2026-09-11): the rebuild standard requires every removal of rows to be
       declared; these two operators let the declaration be checked. Origin: the DAMA-76 rebuild. #}
    {% elif condition.startswith('<= ') %}
        {% set threshold = condition.replace('<= ', '') | float %}
        {% set passed = (actual <= threshold) %}
    {% elif condition.startswith('< ') %}
        {% set threshold = condition.replace('< ', '') | float %}
        {% set passed = (actual < threshold) %}
    {% elif condition == 'increased' %}
        {% set before_val = run_query("
            SELECT CAST(metric_value AS FLOAT) FROM " ~ ls ~ ".CONFIG.BUILD_BASELINE
            WHERE table_name = '" ~ tbl ~ "' AND metric_name = '" ~ metric ~ "'
            AND snapshot_type = 'before' AND database_name = '" ~ db ~ "'
        ") %}
        {% if before_val and before_val.rows | length > 0 %}
            {% set passed = (actual > before_val.rows[0][0]) %}
        {% else %}
            {% set passed = (actual > 0) %}
        {% endif %}
    {% elif condition == 'unchanged' %}
        {% set before_val = run_query("
            SELECT CAST(metric_value AS FLOAT) FROM " ~ ls ~ ".CONFIG.BUILD_BASELINE
            WHERE table_name = '" ~ tbl ~ "' AND metric_name = '" ~ metric ~ "'
            AND snapshot_type = 'before' AND database_name = '" ~ db ~ "'
        ") %}
        {% if before_val and before_val.rows | length > 0 %}
            {% set passed = (actual == before_val.rows[0][0]) %}
        {% else %}
            {% set passed = false %}
        {% endif %}
    {% endif %}

    {% set actual_str = actual | string if actual is not none else "NOT FOUND" %}

    {% if passed %}
        {{ log("  PASS [" ~ version_tag ~ "] " ~ tbl ~ " | " ~ metric ~ " | " ~ condition ~ " | actual: " ~ actual_str ~ " | " ~ description, info=True) }}
    {% else %}
        {{ log("  FAIL [" ~ version_tag ~ "] " ~ tbl ~ " | " ~ metric ~ " | " ~ condition ~ " | actual: " ~ actual_str ~ " | " ~ description, info=True) }}
        {% do failures.append("EXPECTATION: [" ~ version_tag ~ "] " ~ tbl ~ " " ~ metric ~ " expected " ~ condition ~ " but got " ~ actual_str) %}
    {% endif %}
{% endfor %}

{# ================================================================ #}
{# VERDICT                                                          #}
{# ================================================================ #}

{{ log("", info=True) }}
{{ log("--- VERDICT ---", info=True) }}

{% if failures | length > 0 %}
    {{ log("", info=True) }}
    {{ log("WARNINGS — " ~ failures | length ~ " issue(s):", info=True) }}
    {% for f in failures %}
        {{ log("  " ~ f, info=True) }}
    {% endfor %}
    {{ log("", info=True) }}
    {{ log("Build did not stop. Review warnings; the goal is a lookout build (zero warnings).", info=True) }}
    {{ log("", info=True) }}
{% else %}
    {{ log("", info=True) }}
    {{ log("LOOKOUT — all guardrails and expectations met. Zero warnings.", info=True) }}
    {{ log("", info=True) }}
{% endif %}

{%- endmacro -%}
